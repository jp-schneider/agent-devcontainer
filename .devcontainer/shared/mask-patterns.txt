# Files hidden from the container: /dev/null is mounted read-only over every matching file
# (see protect-paths.sh). The host copy is untouched.
#
# One POSIX extended regex (grep -E) per line, matched against the path relative to the
# workspace root (e.g. "sub/dir/.env"). Lines starting with "!" are allow patterns and win
# over everything else. Lines starting with "#" are comments.
# Project-specific additions/exceptions: .devcontainer/mask-patterns.txt (same format).
# .git/ and node_modules/ are not scanned.

# --- Env files: host-only. container.env is the file meant for the container.
(^|/)\.env(\..+)?$
(^|/)[^/]+\.env$
!^container\.env$
!(^|/)\.env\.(example|sample|template)$

# --- Private keys and certificates with keys
\.(pem|key|p12|pfx|jks|keystore|kdbx)$
(^|/)id_(rsa|dsa|ecdsa|ed25519)$

# --- Credential and token files
(^|/)\.netrc$
(^|/)\.npmrc$
(^|/)\.pypirc$
(^|/)\.git-credentials$
(^|/)\.htpasswd$
(^|/)\.aws/(credentials|config)$
(^|/)\.docker/config\.json$
(^|/)credentials(\.(json|ya?ml|toml|ini|txt|cfg|conf))?$
(^|/)secrets?(\.(json|ya?ml|toml|ini|txt|cfg|conf))?$
(^|/)[^/]*service[-_]?account[^/]*\.json$

# --- Infrastructure state with embedded secrets
\.tfvars$
(^|/)terraform\.tfstate(\.backup)?$
